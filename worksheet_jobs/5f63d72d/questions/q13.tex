\begin{question}
\noindent A space agency is testing a circular parabolic mirror segment for a new orbital telescope. During calibration, a laser guide beam is fired so that it just grazes the edge of the circular mirror rim at point $C$, touching it at exactly one point (i.e. the beam is a tangent to the circle at $C$).

Points $A$, $B$ and $C$ lie on the circumference of the mirror's circular cross-section. The chord $CB$ is a support strut inside the mirror housing, and chord $CA$ is a second strut. The laser beam $TCM$ is tangent to the circle at $C$.

The engineers measure the angle between the laser beam and strut $CB$, on the side nearer $M$, to be $64^\circ$. They also know that angle $BAC = 82^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2.2]
    % mark points on circle
        \coordinate (E) at (0,0);
        \coordinate (C) at (-90:1);
        \coordinate (A) at (150:1);

    % draw circle
        \draw[name path=circ] (0,0) circle (1);

    % draw tangent line through C
        \draw (C) -- ($(C)!1.6!-90:(E)$) coordinate (M);
        \draw (C) -- ($(C)!1.6!90:(E)$) coordinate (T);

    % chord CB direction and intersection with circle
        \path[name path=CB] (C) -- ($(C)!1.8!-40:(E)$);
        \path[name intersections={of=CB and circ, by={temp,B}}];

    \draw (C) -- (B);
    \draw (C) -- (A);
    \draw (A) -- (B);
    \draw (A) -- (C);

    % Labels
    \node at (A) [anchor=east] {$A$};
    \node at (B) [anchor=west] {$B$};
    \node at (C) [anchor=north] {$C$};
    \node at (M) [anchor=north] {$M$};
    \node at (T) [anchor=south] {$T$};

    \pic [draw, "$64^\circ$", angle eccentricity=1.5, angle radius = 0.6cm] {angle = B--C--M};
    \pic [draw, "$82^\circ$", angle eccentricity=1.5, angle radius = 0.5cm] {angle = C--A--B};

\end{tikzpicture}
\end{center}

\begin{enumerate}
    \item[(a)] Calculate the size of angle $BCA$.
    \vspace{3.5cm}
    \answereq{\angle BCA}{2}

    \item[(b)] Hence calculate the size of angle $CBA$, the angle at strut junction $B$.
    \vspace{3.5cm}
    \answereq{\angle CBA}{2}
\end{enumerate}
\end{question}
