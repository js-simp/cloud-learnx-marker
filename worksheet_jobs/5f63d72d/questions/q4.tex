\begin{question}
\noindent $A$, $B$, $C$ and $D$ are points on the circumference of a circle, as shown in the diagram below. Chord $AB$ is subtended by angles at points $C$ and $D$ on the circumference.

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate (A) at (140:1);
    \coordinate (B) at (20:1);
    \coordinate (C) at (250:1);
    \coordinate (D) at (300:1);

    \draw[name path=circ] (0,0) circle (1);

    \draw (A) -- (B) -- (C) -- cycle;
    \draw (A) -- (D) -- (B);

    \node at (A) [above left] {$A$};
    \node at (B) [above right] {$B$};
    \node at (C) [below left] {$C$};
    \node at (D) [below right] {$D$};

    \pic [draw, "$x$", angle eccentricity=1.4, angle radius = 0.4cm] {angle = B--C--A};
    \pic [draw, "$47^\circ$", angle eccentricity=1.4, angle radius = 0.4cm] {angle = B--D--A};
\end{tikzpicture}
\end{center}

\noindent Angle $ADB = 47^\circ$.

\noindent Find the size of angle $ACB$, marked $x$ on the diagram.

\vspace{4cm}

\begin{flushright}
\answerequnit{x}{$^\circ$}{2}
\end{flushright}
\end{question}
