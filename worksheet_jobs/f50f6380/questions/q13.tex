    \def\rectwidth{4}

    \node at (0.7*\rectwidth+0.9, \rectheight/2 +0.3) {};
\end{tikzpicture}
\end{center}

\noindent The plinth consists of a rectangle with width $2x$ cm and height $(x+4)$ cm, joined to a right-angled triangle with base $x$ cm and height $(x+4)$ cm, as shown.

\noindent Show that the total area, $A$ cm$^2$, of the plinth is given by
