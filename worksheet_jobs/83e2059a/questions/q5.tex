\begin{question}
\noindent An astronomer is sketching a distant nebula as seen through a telescope. The lens produces a magnified image of shape $T$ shown on the grid below, enlarged from a centre of enlargement $C$.

\begin{center}
\begin{tikzpicture}[scale=0.6, transform shape]
    \definecolor{borderColour}{HTML}{000000}
    \definecolor{fillColour}{HTML}{D8D8D8}

    \def\axisLimit{8}

    % Draw grid
    \draw[step=1cm,gray,very thin] (-\axisLimit,-3) grid (\axisLimit,\axisLimit);

    % Draw axes
    \draw[thick,->] (-\axisLimit-0.5,0) -- (\axisLimit+0.5,0) node[anchor=north] {$x$};
    \draw[thick,->] (0,-3.5) -- (0,\axisLimit+0.5) node[anchor=east] {$y$};

    % Label x-axis and y-axis
    \foreach \x in {-8,-6,-4,-2,2,4,6,8}
        \draw (\x cm,1pt) -- (\x cm,-1pt) node[anchor=north] {$\x$};
    \foreach \y in {-2,2,4,6,8}
        \draw (1pt,\y cm) -- (-1pt,\y cm) node[anchor=east] {$\y$};

    % Centre of enlargement
    \coordinate (C) at (-6,-2);
    \draw[red, thick] (C) node[cross out, draw=red, inner sep=2pt, outer sep=0pt] {};
    \node[below] at (-6,-2.3) {$C$};

    % Define the vertices for shape T
    \coordinate (A) at (-4,1);
    \coordinate (B) at (-4,3);
    \coordinate (D) at (-2,1);

    % Draw shape T
    \draw[fill=fillColour, draw=borderColour, line width=1pt] (A) -- (B) -- (D) -- cycle;
    \node[left] at (-4,2) {$T$};

\end{tikzpicture}
\end{center}

\noindent Enlarge shape $T$ by scale factor $2$, using $C$ as the centre of enlargement.

\vspace{6cm}
\begin{flushright}
    \answerplain{3}
\end{flushright}
\end{question}
