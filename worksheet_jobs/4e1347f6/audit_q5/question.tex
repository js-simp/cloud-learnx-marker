\begin{question}
\noindent A satellite ground station uses a circular dish reflector to receive signals from a spacecraft during a deep-space mission. The signal path from the receiver arm meets the edge of the circular reflector at point $T$, touching it at exactly one point, so that the receiver arm $ST$ is a tangent to the circle with centre $O$.

$OT$ is a radius of the reflector, and $OS$ is a support strut connecting the centre of the dish to a fixing point $S$ on the receiver arm. The angle between the support strut $OS$ and the radius $OT$ is $34^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2.2]
    % centre and tangent point
    \coordinate (O) at (0,0);
    \coordinate (T) at (1,0);

    % circle
    \draw[name path=circ] (O) circle (1);
    \fill[black] (O) circle (0.5pt);
    \node[left] at (O) {$O$};

    % tangent line at T (vertical, perpendicular to radius OT)
    \draw (T) -- ++(0,1.3) coordinate (Ttop);
    \draw (T) -- ++(0,-0.9) coordinate (Tbot) node[below]{$S$};
    \node[right] at (Ttop) {};

    % label T
    \node[below right] at (T) {$T$};

    % radius OT
    \draw (O) -- (T);

    % strut OS
    \coordinate (S) at (Tbot);
    \draw (O) -- (S);

    % right angle mark at T between radius and tangent
    \tkzMarkRightAngle[size=.18](O,T,Tbot);

    % angle at O between OT and OS
    \pic [draw, "$34^\circ$", angle eccentricity=1.5, angle radius=0.55cm] {angle = T--O--S};

    % angle to find at S
    \node at (0.85,-0.75) {$x^\circ$};
\end{tikzpicture}
\end{center}

\noindent Work out the size of angle $OST$, marked $x^\circ$ on the diagram, giving a reason for each stage of your working.

\vspace{5cm}

\begin{flushright}
\answerequnit{x}{$^\circ$}{3}
\end{flushright}
\end{question}
