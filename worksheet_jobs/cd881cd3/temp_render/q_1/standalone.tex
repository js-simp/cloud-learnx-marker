\begin{question}
\noindent The diagram shows a circle with centre $O$. Five features of the circle have been labelled with the letters $P$, $Q$, $R$, $S$ and $T$.

\begin{center}
\begin{tikzpicture}[scale=2.2]
    % centre
    \coordinate (O) at (0,0);
    \draw[thick] (O) circle (1);
    \fill[black] (O) circle (0.4pt);

    % Radius: from centre to a point on circumference
    \coordinate (Rpt) at (35:1);
    \draw[thick] (O) -- (Rpt);
    \node[label={[label distance=1pt]80:$P$}] at ($(O)!0.5!(Rpt)$) {};

    % Diameter: full line through centre
    \coordinate (Dleft) at (200:1);
    \coordinate (Dright) at (20:1);
    \draw[thick] (Dleft) -- (Dright);
    \node at ($(O)!0.6!(200:1)$) [above] {$Q$};

    % Chord: line joining two points on circumference, not through centre
    \coordinate (Ca) at (100:1);
    \coordinate (Cb) at (170:1);
    \draw[thick] (Ca) -- (Cb);
    \node at ($(Ca)!0.5!(Cb)$) [above left] {$R$};

    % Tangent: line touching circle at one point, outside
    \coordinate (Tpt) at (270:1);
    \draw[thick] ($(Tpt)+(-0.9,0)$) -- ($(Tpt)+(0.9,0)$);
    \node at (-0.55,-1.25) {$S$};

    % Arc: mark a short arc on the circumference between two points
    \draw[thick, red] (280:1) arc (280:330:1);
    \node at (305:1.18) {$T$};

\end{tikzpicture}
\end{center}

\noindent Using the words \textbf{radius}, \textbf{diameter}, \textbf{chord}, \textbf{tangent} and \textbf{arc}, write down the mathematical name of the parts labelled $P$, $Q$, $R$, $S$ and $T$.
