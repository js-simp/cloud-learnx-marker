\noindent An engineer is designing a satellite dish. The cross-section of the dish's circular rim, together with some key lines used in the design, is shown in the diagram below.

\begin{center}
\begin{tikzpicture}[scale=2]
    % centre of circle
    \coordinate (O) at (0,0);

    % points on circle
    \coordinate (P) at (100:1);
    \coordinate (Q) at (170:1);
    \coordinate (R) at (10:1);
    \coordinate (S) at (-40:1);

    % draw circle
    \draw[thick] (O) circle (1);
    \fill (O) circle (0.6pt);

    % Line 1: centre to circumference (radius) O to P
    \draw[thick] (O) -- (P);

    % Line 2: chord Q to R (does not pass through centre)
    \draw[thick] (Q) -- (R);

    % Line 3: tangent at S
    \draw[thick] ($(S)!1.3!90:(O)$) -- ($(S)!1.3!-90:(O)$);

    % Line 4: arc between Q and R marked with a thicker arc (highlight minor arc QR)
    \draw[thick, red] (Q) arc (170:190:1);

    % Labels for the four features
    \node at ($(O)!0.5!(P)$) [above right] {$A$};
    \node at ($(Q)!0.5!(R)$) [above] {$B$};
    \node at (S) [below] {$C$};
    \node[red] at (180:1.12) {$D$};

    % dot at centre label
    \node at (O) [below right] {$O$};

\end{tikzpicture}
\end{center}

\noindent The circle represents the outer rim of the dish, and $O$ is the centre.

\noindent Line $A$, line $B$, line $C$, and the highlighted curve $D$ (shown in red) each represent a different geometric feature of a circle.

\noindent Match each letter to the correct term from the list below:

\begin{center}
\textbf{radius} \quad \textbf{chord} \quad \textbf{tangent} \quad \textbf{arc}
\end{center}
