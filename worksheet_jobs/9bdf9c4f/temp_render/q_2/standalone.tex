\begin{question}
\noindent $A$, $B$ and $C$ are points on the circumference of a circle with centre $O$.
$AC$ is a diameter of the circle.

\begin{center}
\begin{tikzpicture}[scale=2.2]
    \coordinate (O) at (0,0);
    \coordinate (A) at (180:1);
    \coordinate (C) at (0:1);
    \coordinate (B) at (55:1);

    \draw[thick] (0,0) circle (1);
    \fill[black] (O) circle (0.6pt);

    \draw[thick] (A) -- (C);
    \draw[thick] (A) -- (B);
    \draw[thick] (B) -- (C);

    \node[left] at (A) {$A$};
    \node[right] at (C) {$C$};
    \node[above] at (B) {$B$};
    \node[below] at (O) {$O$};

    \node at (0.05, 0.15) {$68^\circ$};
    \pic [draw, angle eccentricity=1.5, angle radius=0.35cm, "" ] {angle = C--A--B};

    \node at (0.75,0.3) {$x$};
\end{tikzpicture}
\end{center}

\noindent Angle $BAC = 68^\circ$.

\noindent Work out the size of angle $x$, marked at $B$ in the diagram.

\vspace{4cm}

\begin{flushright}
\answerequnit{x}{$^\circ$}{2}
\end{flushright}
\end{question}
