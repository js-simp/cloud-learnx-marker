\begin{question}
\noindent $A$, $B$ and $C$ are points on the circumference of a circle with centre $O$.

\noindent Angle $AOC = 80^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2]
    %mark points on circle
        \coordinate[] (O) at (0,0);
        \coordinate[] (A) at (130:1);
        \coordinate[] (C) at (50:1);
        \coordinate[] (B) at (250:1);

    %draw chords and circle
        \draw[name path=circ] (0,0) circle (1);
        \fill[black] (0,0) circle (0.6pt);
        \draw (A) -- (O);
        \draw (O) -- (C);
        \draw (A) -- (B);
        \draw (B) -- (C);

        \node[above left] at (A) {$A$};
        \node[above right] at (C) {$C$};
        \node[below] at (B) {$B$};
        \node[above] at (0,0.15) {$O$};

        \pic [draw, "$80^\circ$", angle eccentricity=1.5, angle radius = 0.5cm] {angle = C--O--A};
        \pic [draw, "$x$", angle eccentricity=1.6, angle radius = 0.5cm] {angle = C--B--A};
        
    \end{tikzpicture}
\end{center}

\noindent Find the size of angle $x$.

\noindent Give a reason for your answer.
\vspace{4cm}

\begin{flushright}
    \answerequnit{x}{$^\circ$}{2}
\end{flushright}
\end{question}
