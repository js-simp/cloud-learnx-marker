\begin{question}
\noindent $A$, $B$ and $C$ are points on the circumference of a circle with centre $O$.

\noindent Find the size of angle $ACB$, marked $x$ on the diagram.

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate (O) at (0,0);
    \coordinate (A) at (140:1);
    \coordinate (B) at (20:1);
    \coordinate (C) at (250:1);

    \draw[name path=circ, fill = orange!10] (0,0) circle (1);
    \fill[black] (0,0) circle (1pt);

    \draw (O) -- (A);
    \draw (O) -- (B);
    \draw (A) -- (C);
    \draw (C) -- (B);

    \node[above left] at (A) {$A$};
    \node[above right] at (B) {$B$};
    \node[below] at (C) {$C$};
    \node[below right] at (O) {$O$};

    \pic [draw, "$100^\circ$", angle eccentricity=1.5, angle radius = 0.5cm] {angle = A--O--B};
    \pic [draw, "$x$", angle eccentricity=1.6, angle radius = 0.4cm] {angle = B--C--A};
\end{tikzpicture}
\end{center}

\vspace{4cm}
\begin{flushright}
    \answerequnit{x}{$^\circ$}{2}
\end{flushright}
\end{question}
