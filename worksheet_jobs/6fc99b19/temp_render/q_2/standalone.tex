\begin{question}
\noindent $A$, $B$ and $C$ are points on the circumference of a circle with centre $O$.
$AC$ is a diameter of the circle.

\begin{center}
\begin{tikzpicture}[scale=2.4]
    \coordinate (O) at (0,0);
    \coordinate (A) at (180:1);
    \coordinate (C) at (0:1);
    \coordinate (B) at (55:1);

    \draw[fill=primaryGreen!20] (O) circle (1);
    \fill[black] (O) circle (0.6pt);

    \draw[thick] (A) -- (C);
    \draw[thick] (A) -- (B) -- (C);

    \tkzMarkRightAngle[size=.18](A,B,C);

    \node[left] at (A) {$A$};
    \node[right] at (C) {$C$};
    \node[above right] at (B) {$B$};
    \node[below] at (O) {$O$};

    \node at ($(A)!0.35!(B) + (0.05,0.06)$) {$27^\circ$};
    \node at ($(B)!0.5!(C) + (0.02,0.08)$) {$x$};
\end{tikzpicture}
\end{center}

\noindent Angle $BAC = 27^\circ$.

Work out the size of angle $BCA$, marked $x$ on the diagram.

\vspace{4cm}

\begin{flushright}
    \answerequnit{x}{$^\circ$}{2}
\end{flushright}
\end{question}
