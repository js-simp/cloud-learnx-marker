\begin{question}
\noindent A laboratory supplier makes two mathematically similar conical flasks, a small one and a large one.

\noindent The volume of the small flask is $64 \text{ cm}^3$.\\
The volume of the large flask is $1728 \text{ cm}^3$.

\noindent The height of the small flask is $6$ cm.

\noindent Work out the height of the large flask.

\vspace{5cm}

\begin{flushright}
    \answerunit{cm}{4}
\end{flushright}
\end{question}
