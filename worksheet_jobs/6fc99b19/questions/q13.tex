\begin{question}
\noindent A telescope manufacturer produces two similar circular mirrors, a small mirror and a large mirror, for two different telescope models.

\noindent The diameter of the small mirror is $12$ cm and the diameter of the large mirror is $18$ cm.

\noindent The surface area of the small mirror is $113.1$ cm$^2$.

\noindent Calculate the surface area of the large mirror.

\vspace{5cm}

\begin{flushright}
    \answerunit{$\text{cm}^2$}{3}
\end{flushright}
\end{question}
