\begin{question}
\noindent Triangle $S$ is shown on the grid below.

\begin{center}
\begin{tikzpicture}[scale=0.75, transform shape]
    \definecolor{borderColour}{HTML}{000000}
    \definecolor{fillColourS}{HTML}{ADD8E6}

    \def\axisLimit{8}

    % Draw grid
    \draw[step=1cm,gray,very thin] (-1,-1) grid (\axisLimit,\axisLimit);

    % Draw axes
    \draw[thick,->] (-1.5,0) -- (\axisLimit+0.5,0) node[anchor=north] {$x$};
    \draw[thick,->] (0,-1.5) -- (0,\axisLimit+0.5) node[anchor=east] {$y$};

    % Label x-axis and y-axis
    \foreach \x in {0,1,...,\axisLimit}
        \draw (\x cm,1pt) -- (\x cm,-1pt) node[anchor=north] {$\x$};
    \foreach \y in {0,1,...,\axisLimit}
        \draw (1pt,\y cm) -- (-1pt,\y cm) node[anchor=east] {$\y$};

    % Centre of enlargement
    \coordinate (O) at (1,1);
    \draw[red, thick] (O) node[cross out, draw=red, inner sep=2pt, outer sep=0pt, line width=1pt] {};
    \node[below left] at (O) {$O$};

    % Define vertices of S
    \coordinate (A) at (2,2);
    \coordinate (B) at (2,4);
    \coordinate (C) at (3,2);

    % Draw S
    \draw[fill=fillColourS, draw=borderColour, line width=1pt] (A) -- (B) -- (C) -- cycle;
    \node[right] at (2.6,2.9) {$S$};

    \node[below right] at (A) {$A$};
    \node[left] at (B) {$B$};
    \node[below right] at (C) {$C$};

\end{tikzpicture}
\end{center}

\noindent Enlarge triangle $S$ by scale factor $2$, using $O$ as the centre of enlargement.\\
Label the image $S'$.

    \answerlines{2}{3}
\end{flushright}
\end{question}
