\begin{question}
\noindent An e-sports tournament organiser is setting up three signal relay towers, $A$, $B$ and $C$, to broadcast a live Valorant tournament across a stadium complex. Tower $B$ is 15 km from tower $A$, on a bearing of $048^\circ$ from $A$. Tower $C$ is 21 km from tower $B$, on a bearing of $112^\circ$ from $B$.

\begin{center}
\begin{tikzpicture}[scale=0.9, transform shape]
    \coordinate (A) at (0,0);
    % Bearing of B from A is 48, angle from horizontal (east) = 90-48 = 42
    \coordinate (B) at (42:6.5);
    % Bearing of C from B is 112. Direction line BC makes with North-at-B: 112 deg clockwise from north.
    % angle from horizontal at B = 90-112 = -22
    \coordinate (C) at ($(B) + (-22:8.5)$);

    % North line at A
    \draw[dashed] (A) -- (0,3.2);
    \draw[->] (0,2.9) -- (0,3.2);
    \node at (0,3.4) [font=\small] {N};

    % North line at B
    \draw[dashed] (B) -- ($(B)+(0,3)$);
    \draw[->] ($(B)+(0,2.7)$) -- ($(B)+(0,3)$);
    \node at ($(B)+(0,3.2)$) [font=\small] {N};

    \coordinate (JA) at (0,3.2);
    \coordinate (JB) at ($(B)+(0,3)$);

    % Bearing angle at A
    \pic [draw, angle radius=1cm, "$48^\circ$", angle eccentricity=1.4] {angle = B--A--JA};

    % Bearing angle at B (from north line at B to C, clockwise)
    \pic [draw, angle radius=1cm, "$112^\circ$", angle eccentricity=1.4] {angle = C--B--JB};

    % Triangle sides
    \draw[thick] (A) -- (B) node[midway, above left, font=\small] {$15$ km};
    \draw[thick] (B) -- (C) node[midway, above right, font=\small] {$21$ km};
    \draw[thick] (A) -- (C) node[midway, below, font=\small] {$d$ km};

    \node at (A) [below left, font=\small] {$A$};
    \node at (B) [above, font=\small] {$B$};
    \node at (C) [right, font=\small] {$C$};
\end{tikzpicture}
\end{center}

\begin{enumerate}
    \item[(a)] Work out the size of angle $ABC$.

    \vspace{4cm}
    \answerequnit{ABC}{$^\circ$}{2}

    \item[(b)] Hence work out the distance $AC$, giving your answer to 3 significant figures.

    \vspace{4.5cm}
    \answerunit{km}{3}
\end{enumerate}
\end{question}
