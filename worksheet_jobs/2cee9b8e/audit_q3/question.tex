% Continuous exam box with internal dividers
\newcommand{\examboxes}[1]{%
\begin{tikzpicture}[baseline=(current bounding box.center)]
    % dimensions
    \def\boxwidth{0.9}   % width of each small box
    \def\boxheight{1}  % height
    \def\totalwidth{#1*\boxwidth}
    
    % outer rounded rectangle
    \draw[line width=2pt, rounded corners=3pt, color=black!60]
        (0,0) rectangle (\totalwidth,\boxheight);
    
    % internal vertical lines
    \foreach \i in {1,...,#1} {
        \ifnum\i<#1
            \draw[line width=2pt, color=black!60]
            (\i*\boxwidth,0) -- (\i*\boxwidth,\boxheight);
        \fi
    }
\end{tikzpicture}
}

% answer marks
\newlength{\answerwidth}
\setlength{\answerwidth}{3cm}
% Core right-aligned answer block
\newcommand{\answerbox}[3]{%
    % #1 = prefix
    % #2 = suffix
    % #3 = marks
    \vspace{2mm}
    \noindent\hfill
    \begin{minipage}{4.5cm}
        \raggedleft
        #1\makebox[\answerwidth]{\dotfill}#2
        \vspace{1mm}
        
        \hfill\makebox[\answerwidth][c]{\textbf{(#3)}}
    \end{minipage}
    \addtocounter{questionmarks}{#3}
    \addtocounter{totalmarks}{#3}
}

% 1) Just dotted line
\newcommand{\answerplain}[1]{%
    \answerbox{}{}{#1}
}

% 2) Dotted line + unit after
\newcommand{\answerunit}[2]{%
    \answerbox{}{\,#1}{#2}
}

% 3) Unit before dotted line
\newcommand{\answerprefix}[2]{%
    \answerbox{#1\,}{}{#2}
}

% 4) Variable before dotted line (e.g. x = )
\newcommand{\answereq}[2]{%
    \answerbox{$#1 =$\,}{}{#2}
}

% 5) Variable + unit
\newcommand{\answerequnit}[3]{%
    \answerbox{$#1 =$\,}{\,#2}{#3}
}

% Explanation / reasoning space
\newcommand{\answerlines}[2]{%
    % #1 = number of lines
    % #2 = marks
    
    \vspace{2mm}
    
    \noindent
    \begin{minipage}{\linewidth}
        \foreach \i in {1,...,#1} {
            \noindent\hdashrule[0.5ex]{\linewidth}{0.5pt}{2pt}\\[3mm]
        }
        
        \vspace{1mm}
        \noindent\hfill\textbf{(#2)}
    \end{minipage}
    
    \addtocounter{questionmarks}{#2}
    \addtocounter{totalmarks}{#2}
}

\newcommand{\answermarks}[1]{%

    \addtocounter{totalmarks}{#1}
}

\begin{question}
\noindent Points $A$, $B$, $C$ and $D$ lie on the circumference of a circle, as shown in the diagram. Chord $AD$ is drawn, and points $B$ and $C$ lie on the same side of $AD$.

Angle $ABD = 38^\circ$ and angle $BDC = 24^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2.4]
    \coordinate (A) at (200:1);
    \coordinate (D) at (-20:1);
    \coordinate (B) at (100:1);
    \coordinate (C) at (60:1);

    \draw[name path=circ] (0,0) circle (1);

    \draw (A) -- (D);
    \draw (A) -- (B) -- (D);
    \draw (A) -- (C) -- (D);

    \node[left] at (A) {$A$};
    \node[right] at (D) {$D$};
    \node[above] at (B) {$B$};
    \node[above right] at (C) {$C$};

    \pic [draw, angle eccentricity=1.5, angle radius = 0.35cm, "$38^\circ$"] {angle = D--B--A};
    \pic [draw, angle eccentricity=1.5, angle radius = 0.3cm, "$24^\circ$"] {angle = C--D--B};

\end{tikzpicture}
\end{center}

\noindent Given that angle $ABD = 38^\circ$ and angle $BDC = 24^\circ$, find the size of angle $ACD$, giving reasons for your answer.

\vspace{4cm}
\begin{flushright}
    \answerequnit{ACD}{$^\circ$}{3}
\end{flushright}
\end{question}
