\begin{question}
\noindent A space agency uses a circular satellite dish, shown below. Points $A$, $B$ and $C$ lie on the circumference of the circle with centre $O$, and $AC$ is a diameter of the dish.

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate (O) at (0,0);
    \coordinate (A) at (180:1);
    \coordinate (C) at (0:1);
    \coordinate (B) at (55:1);

    \draw (O) circle (1);
    \fill[black] (O) circle (0.5pt);

    \draw[thick] (A) -- (C);
    \draw[thick] (A) -- (B);
    \draw[thick] (B) -- (C);

    \node[left] at (A) {$A$};
    \node[right] at (C) {$C$};
    \node[above] at (B) {$B$};
    \node[below] at (O) {$O$};

    \pic [draw, angle radius=6mm, "$x$"] {angle = C--B--A};
\end{tikzpicture}
\end{center}

\noindent Find the size of angle $x$, giving a reason for your answer.

\vspace{4cm}

\begin{flushright}
\answerequnit{x}{$^\circ$}{2}
\end{flushright}
\end{question}
