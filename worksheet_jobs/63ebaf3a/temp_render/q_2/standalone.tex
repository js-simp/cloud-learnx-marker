\begin{question}
\noindent $A$, $B$ and $C$ are points on the circumference of a circle with centre $O$.

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate (O) at (0,0);
    \coordinate (A) at (130:1);
    \coordinate (B) at (-30:1);
    \coordinate (C) at (260:1);

    \draw[name path=circ, fill=orange!15] (0,0) circle (1);
    \fill[black] (O) circle (0.6pt);
    \draw (A) -- (O);
    \draw (O) -- (B);
    \draw (A) -- (C);
    \draw (C) -- (B);

    \node[above left] at (A) {$A$};
    \node[right] at (B) {$B$};
    \node[below] at (C) {$C$};
    \node[below right] at (O) {$O$};

    \pic [draw, "$x$", angle eccentricity=1.5, angle radius=0.5cm] {angle = A--O--B};
    \pic [draw, "$52^\circ$", angle eccentricity=1.4, angle radius=0.5cm] {angle = A--C--B};
\end{tikzpicture}
\end{center}

\noindent Angle $ACB = 52^\circ$.

\noindent Find the size of angle $x$, the angle at the centre of the circle.
State the circle theorem you have used.

\vspace{4cm}

\begin{flushright}
\answerequnit{x}{$^\circ$}{2}
\end{flushright}
\end{question}
