\begin{question}
\noindent Shape $P$ (an esports team's logo, shaped like an arrow) is translated to shape $Q$, as shown on the grid below.

\begin{center}
\begin{tikzpicture}[scale=0.7]
    % Define colours
    \definecolor{borderColour}{HTML}{000000}
    \definecolor{fillColourP}{HTML}{D8D8D8}
    \definecolor{fillColourQ}{HTML}{B8D8FF}

    % Define axis scale
    \def\axisLimit{6}

    % Draw the grid
    \draw[step=1cm,gray,very thin] (-\axisLimit,-\axisLimit) grid (\axisLimit,\axisLimit);
    \draw[thick,->] (-\axisLimit-0.5,0) -- (\axisLimit+0.5,0) node[anchor=north] {$x$};
    \draw[thick,->] (0,-\axisLimit-0.5) -- (0,\axisLimit+0.5) node[anchor=east] {$y$};

    % Label x-axis and y-axis, skipping 0
    \foreach \x in {-6,-5,...,-1,1,2,...,6}
        \draw (\x cm,1pt) -- (\x cm,-1pt) node[anchor=north] {$\x$};
    \foreach \y in {-6,-5,...,-1,1,2,...,6}
        \draw (1pt,\y cm) -- (-1pt,\y cm) node[anchor=east] {$\y$};

    % Define vertices of shape P (an arrow-like quadrilateral)
    \coordinate (A) at (-5,-4);
    \coordinate (B) at (-5,-1);
    \coordinate (C) at (-3,-1);
    \coordinate (D) at (-3,-4);

    % Draw shape P
    \filldraw[fill=fillColourP, draw=borderColour, line width=1pt] (A) -- (B) -- (C) -- (D) -- cycle;

    % Define vertices of shape Q (translated by vector (5,6))
    \coordinate (A') at ($(A)+(5,6)$);
    \coordinate (B') at ($(B)+(5,6)$);
    \coordinate (C') at ($(C)+(5,6)$);
    \coordinate (D') at ($(D)+(5,6)$);

    % Draw shape Q
    \filldraw[fill=fillColourQ, draw=borderColour, line width=1pt] (A') -- (B') -- (C') -- (D') -- cycle;

    % Labels
    \node at (-4,-2.5) {$P$};
    \node at (1,3.5) {$Q$};

\end{tikzpicture}
\end{center}

\noindent Work out the translation that maps shape $P$ onto shape $Q$.

\end{question}
