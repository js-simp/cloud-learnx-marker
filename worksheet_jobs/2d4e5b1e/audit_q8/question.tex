% Continuous exam box with internal dividers
\newcommand{\examboxes}[1]{%
\begin{tikzpicture}[baseline=(current bounding box.center)]
    % dimensions
    \def\boxwidth{0.9}   % width of each small box
    \def\boxheight{1}  % height
    \def\totalwidth{#1*\boxwidth}
    
    % outer rounded rectangle
    \draw[line width=2pt, rounded corners=3pt, color=black!60]
        (0,0) rectangle (\totalwidth,\boxheight);
    
    % internal vertical lines
    \foreach \i in {1,...,#1} {
        \ifnum\i<#1
            \draw[line width=2pt, color=black!60]
            (\i*\boxwidth,0) -- (\i*\boxwidth,\boxheight);
        \fi
    }
\end{tikzpicture}
}

% answer marks
\newlength{\answerwidth}
\setlength{\answerwidth}{3cm}
% Core right-aligned answer block
\newcommand{\answerbox}[3]{%
    % #1 = prefix
    % #2 = suffix
    % #3 = marks
    \vspace{2mm}
    \noindent\hfill
    \begin{minipage}{4.5cm}
        \raggedleft
        #1\makebox[\answerwidth]{\dotfill}#2
        \vspace{1mm}
        
        \hfill\makebox[\answerwidth][c]{\textbf{(#3)}}
    \end{minipage}
    \addtocounter{questionmarks}{#3}
    \addtocounter{totalmarks}{#3}
}

% 1) Just dotted line
\newcommand{\answerplain}[1]{%
    \answerbox{}{}{#1}
}

% 2) Dotted line + unit after
\newcommand{\answerunit}[2]{%
    \answerbox{}{\,#1}{#2}
}

% 3) Unit before dotted line
\newcommand{\answerprefix}[2]{%
    \answerbox{#1\,}{}{#2}
}

% 4) Variable before dotted line (e.g. x = )
\newcommand{\answereq}[2]{%
    \answerbox{$#1 =$\,}{}{#2}
}

% 5) Variable + unit
\newcommand{\answerequnit}[3]{%
    \answerbox{$#1 =$\,}{\,#2}{#3}
}

    \addtocounter{totalmarks}{#2}
}

\begin{question}
\noindent $A$, $B$, $C$ and $D$ are four points on the circumference of a circle, forming the cyclic quadrilateral $ABCD$.

\noindent The angle at $A$ is $(2x+10)^\circ$ and the angle at $C$ is $(3x-30)^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate[label=above left:$B$] (B) at (130:1);
    \coordinate[label=below left:$A$] (A) at (215:1);
    \draw[name path=circ] (0,0) circle (1);
    \path[name path=BD] (B) -- ($(B)!1.5!66:(A)$);
    \path[name path=AC] (A) -- ($(A)!1.5!-50:(B)$);
    \path[name intersections={of=circ and BD, by={[]K,D}}];
    \path[name intersections={of=AC and circ, by={[label=above right:$C$]C}}];
    \draw[line join=bevel] (A) -- (B) -- (D) -- (C) -- cycle;
    \node at (D) [right] {$D$};

    \pic [draw, "$(2x+10)^\circ$", angle eccentricity=1.6, angle radius = 0.5cm] {angle = D--A--B};
    \pic [draw, "$(3x-30)^\circ$", angle eccentricity=1.6, angle radius = 0.4cm] {angle = B--C--D};
\end{tikzpicture}
\end{center}

\noindent A historian studying the astrolabe used by medieval astronomers notices that its main dial is divided by a cyclic quadrilateral marked exactly like this one, and wants to check the calibration is correct.

\begin{enumerate}
    \item[(a)] Show that $x = 40$.

    \vspace{4cm}
    \answermarks{2}

    \item[(b)] Find the size of each of the four angles of the cyclic quadrilateral $ABCD$.

    \vspace{5cm}
    \answerplain{2}
\end{enumerate}
\end{question}
