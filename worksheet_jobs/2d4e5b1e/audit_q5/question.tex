% Continuous exam box with internal dividers
\newcommand{\examboxes}[1]{%
\begin{tikzpicture}[baseline=(current bounding box.center)]
    % dimensions
    \def\boxwidth{0.9}   % width of each small box
    \def\boxheight{1}  % height
    \def\totalwidth{#1*\boxwidth}
    
    % outer rounded rectangle
    \draw[line width=2pt, rounded corners=3pt, color=black!60]
        (0,0) rectangle (\totalwidth,\boxheight);
    
    % internal vertical lines
    \foreach \i in {1,...,#1} {
        \ifnum\i<#1
            \draw[line width=2pt, color=black!60]
            (\i*\boxwidth,0) -- (\i*\boxwidth,\boxheight);
        \fi
    }
\end{tikzpicture}
}

% answer marks
\newlength{\answerwidth}
\setlength{\answerwidth}{3cm}
% Core right-aligned answer block
\newcommand{\answerbox}[3]{%
    % #1 = prefix
    % #2 = suffix
    % #3 = marks
    \vspace{2mm}
    \noindent\hfill
    \begin{minipage}{4.5cm}
        \raggedleft
        #1\makebox[\answerwidth]{\dotfill}#2
        \vspace{1mm}
        
        \hfill\makebox[\answerwidth][c]{\textbf{(#3)}}
    \end{minipage}
    \addtocounter{questionmarks}{#3}
    \addtocounter{totalmarks}{#3}
}

% 1) Just dotted line
\newcommand{\answerplain}[1]{%
    \answerbox{}{}{#1}
}

% 2) Dotted line + unit after
\newcommand{\answerunit}[2]{%
    \answerbox{}{\,#1}{#2}
}

% 3) Unit before dotted line
\newcommand{\answerprefix}[2]{%
    \answerbox{#1\,}{}{#2}
}

% 4) Variable before dotted line (e.g. x = )
\newcommand{\answereq}[2]{%
    \answerbox{$#1 =$\,}{}{#2}
}

% 5) Variable + unit
\newcommand{\answerequnit}[3]{%
    \answerbox{$#1 =$\,}{\,#2}{#3}
}

% Explanation / reasoning space
\newcommand{\answerlines}[2]{%
    % #1 = number of lines
    % #2 = marks
    
    \vspace{2mm}
    
    \noindent
    \begin{minipage}{\linewidth}
        \foreach \i in {1,...,#1} {
            \noindent\hdashrule[0.5ex]{\linewidth}{0.5pt}{2pt}\\[3mm]
        }
        
        \vspace{1mm}
        \noindent\hfill\textbf{(#2)}
    \end{minipage}
    
    \addtocounter{questionmarks}{#2}
    \addtocounter{totalmarks}{#2}
}

\begin{question}
\noindent Triangle $ABC$ is drawn on the grid below. Point $E$ marks the centre of enlargement.

\begin{center}
\begin{tikzpicture}[scale=0.6, transform shape]
    \def\axisLimit{6}

    \definecolor{borderColour}{HTML}{000000}
    \definecolor{fillColour}{HTML}{D8D8D8}

    % Draw grid
    \draw[step=1cm,gray,very thin] (-6,-6) grid (6,6);

    % Draw axes
    \draw[thick,->] (-6.5,0) -- (6.5,0) node[anchor=north] {$x$};
    \draw[thick,->] (0,-6.5) -- (0,6.5) node[anchor=east] {$y$};

    % Label x-axis and y-axis
    \foreach \x in {-6,...,-1,1,2,...,6}
        \draw (\x cm,1pt) -- (\x cm,-1pt) node[anchor=north] {\x};
    \foreach \y in {-6,...,-1,1,2,...,6}
        \draw (1pt,\y cm) -- (-1pt,\y cm) node[anchor=east] {\y};

    % Centre of enlargement
    \coordinate (E) at (-4,4);
    \draw[red, thick] (E) node[cross out, draw=red, inner sep=2pt, outer sep=0pt] {};
    \node at (-4.5,4.5) {$E$};

    % Vertices of triangle ABC
    \coordinate (A) at (0,4);
    \coordinate (B) at (0,0);
    \coordinate (C) at (2,0);

    \draw[fill=fillColour, draw=borderColour, line width=1pt] (A) -- (B) -- (C) -- cycle;
    \node at (0.3,4.3) {$A$};
    \node at (-0.4,-0.2) {$B$};
    \node at (2.4,-0.2) {$C$};

\end{tikzpicture}
\end{center}

\noindent Find the coordinates of the image of triangle $ABC$ after an enlargement with scale factor $\dfrac{1}{2}$, centre $E(-4,4)$.

\vspace{4cm}
\answerplain{3}
\end{question}
