\begin{question}
\noindent Two laboratory conical flasks, $A$ and $B$, are mathematically similar in shape.

\noindent The height of flask $A$ is $6$ cm and the height of flask $B$ is $9$ cm.

\noindent The volume of flask $A$ is $80$ cm$^3$.

\begin{center}
\begin{tikzpicture}[scale=0.7]
    % Flask A (smaller)
    \draw (-1.2,6) -- (-0.4,1.2) arc (200:340:0.9 and 0.35) -- (1.2,6);
    \draw (-1.2,6) -- (-1.5,6.6);
    \draw (1.2,6) -- (1.5,6.6);
    \draw (-1.5,6.6) -- (1.5,6.6);
    \node at (0,-0.3) {Flask $A$};
    \draw[<->] (-2.2,1.2) -- (-2.2,6.6);
    \node at (-2.7,3.9) {$6$ cm};

    % Flask B (larger), positioned to the right
    \draw (5.7,9) -- (6.9,0.9) arc (200:340:1.35 and 0.5) -- (8.1,9);
    \draw (5.7,9) -- (5.25,9.9);
    \draw (8.1,9) -- (8.55,9.9);
    \draw (5.25,9.9) -- (8.55,9.9);
    \node at (6.9,-0.3) {Flask $B$};
    \draw[<->] (4.5,0.9) -- (4.5,9.9);
    \node at (4,5.4) {$9$ cm};
\end{tikzpicture}
\end{center}

\noindent Calculate the volume of flask $B$.

\vspace{5cm}

\begin{flushright}
    \answerunit{$\text{cm}^3$}{3}
\end{flushright}
\end{question}
