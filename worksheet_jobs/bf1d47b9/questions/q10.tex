\begin{question}
\noindent A Model United Nations conference is being held at a large venue. Delegate hub $A$ and the plenary hall $B$ are two locations on the conference site.

\noindent The bearing of $A$ from $B$ is $205^\circ$.

\begin{center}
\begin{tikzpicture}[scale=1.1]
    \coordinate (A) at (0,0);
    \coordinate (B) at (0,3.6);

    % North lines at A and B
    \draw[dashed] (A) -- (0,-1.2);
    \draw[->] (0,-0.9) -- (0,-1.2);
    \node at (0,-1.4) [font=\small] {N};

    \draw[dashed] (B) -- (0,5.4);
    \draw[->] (B) -- (0,5.1);
    \node at (0,5.6) [font=\small] {N};

    % Line joining A and B
    \draw[thick] (A) -- (B);

    % Bearing angle at B: from North (up) round to A (down), measured clockwise = 205 degrees
    % North at B points to (0,5.4)+ up direction, i.e. angle 90 from horizontal.
    % A is directly below B, i.e. angle -90 (i.e. 270) from horizontal, measured from B.
    % Bearing of A from B = 205 degrees clockwise from North at B.
    % So mark the angle from North direction (up, i.e. 90 deg heading) clockwise to BA direction.
    \coordinate (Ndir) at (0,5.4);
    \pic [draw, angle radius=1cm, "$205^\circ$", angle eccentricity=1.5] {angle=A--B--Ndir};

    \node at (A) [below,font=\small] {$A$};
    \node at (B) [above right,font=\small] {$B$};
\end{tikzpicture}
\end{center}

\noindent Work out the bearing of $B$ from $A$.

\vspace{4.5cm}

\begin{flushright}
    \answerequnit{}{$^\circ$}{3}
\end{flushright}
\end{question}
