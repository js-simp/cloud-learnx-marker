\begin{question}
\noindent $A$, $B$ and $C$ are points on the circumference of a circle with centre $O$.
$O$ is joined to $A$ and to $B$, forming radii $OA$ and $OB$.

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate (O) at (0,0);
    \coordinate (A) at (112:1);
    \coordinate (B) at (-24:1);
    \coordinate (C) at (200:1);

    \draw[name path=circ] (0,0) circle (1);
    \fill[black] (O) circle (0.4pt);
    \node[below] at (O) {$O$};

    \draw (O) -- (A);
    \draw (O) -- (B);
    \draw (A) -- (C);
    \draw (C) -- (B);

    \node[above right] at (A) {$A$};
    \node[right] at (B) {$B$};
    \node[left] at (C) {$C$};

    \pic [draw, angle eccentricity=1.5, angle radius = 0.45cm, "$136^\circ$"] {angle = A--O--B};
\end{tikzpicture}
\end{center}

\noindent Angle $AOB = 136^\circ$.

\noindent Find the size of angle $ACB$.

\vspace{4.5cm}

\begin{flushright}
\answerequnit{\text{Angle } ACB}{$^\circ$}{3}
\end{flushright}
\end{question}
