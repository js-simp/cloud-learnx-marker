\begin{question}
\noindent $A$, $B$, $C$ and $D$ are four points on the circumference of a circle, forming the cyclic quadrilateral $ABCD$.

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate[label=above left:$A$] (A) at (150:1);
    \coordinate[label=left:$B$] (B) at (215:1);
    \coordinate[label=below right:$C$] (C) at (300:1);
    \coordinate[label=above right:$D$] (D) at (30:1);

    \draw[name path=circ] (0,0) circle (1);
    \draw[line join=bevel, thick] (A) -- (B) -- (C) -- (D) -- cycle;

    \pic [draw, "$3x^\circ$", angle eccentricity=1.6, angle radius = 0.45cm] {angle = D--A--B};
    \pic [draw, "$(x+52)^\circ$", angle eccentricity=1.6, angle radius = 0.45cm] {angle = B--C--D};
\end{tikzpicture}
\end{center}

\noindent Angle $BAD = 3x$ degrees and angle $BCD = (x + 52)$ degrees.

\vspace{0.3cm}
\noindent Work out the value of $x$, and hence find the sizes of angle $BAD$ and angle $BCD$.

\vspace{5cm}

\begin{flushright}
\answerlines{3}{3}
\end{flushright}
\end{question}
