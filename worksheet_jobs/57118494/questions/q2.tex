\begin{question}
\noindent A designer is creating a circular logo for an e-sports team. Points $P$, $Q$ and $R$ lie on the circumference of the circle, and $PR$ is a diameter of the circle.

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate (P) at (180:1);
    \coordinate (R) at (0:1);
    \coordinate (Q) at (55:1);

    \draw[thick] (0,0) circle (1);
    \fill[black] (0,0) circle (0.6pt);

    \draw[thick] (P) -- (R);
    \draw[thick] (P) -- (Q);
    \draw[thick] (Q) -- (R);

    \node[left] at (P) {$P$};
    \node[right] at (R) {$R$};
    \node[above] at (Q) {$Q$};

    \pic [draw, angle radius=6mm, "$35^\circ$"] {angle = R--P--Q};
    \pic [draw, angle radius=5mm, "$x$"] {angle = P--R--Q};
\end{tikzpicture}
\end{center}

\noindent Angle $QPR = 35^\circ$.

\noindent Work out the size of angle $x$, giving a reason for your answer.

\vspace{4cm}

\begin{flushright}
    \answerequnit{x}{$^\circ$}{2}
\end{flushright}
\end{question}
