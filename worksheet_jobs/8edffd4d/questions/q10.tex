\begin{question}
\noindent Quadrilateral $ABCD$ is drawn on the grid below. Point $O$ is marked with a cross.

\begin{center}
\begin{tikzpicture}[scale=0.6, transform shape]
    \definecolor{borderColour}{HTML}{000000}
    \definecolor{fillColour}{HTML}{D8D8D8}

    \def\axisLimit{8}

    % Draw grid
    \draw[step=1cm,gray,very thin] (-2,-8) grid (8,4);

    % Draw axes
    \draw[thick,->] (-2.5,0) -- (8.5,0) node[anchor=north] {$x$};
    \draw[thick,->] (0,-8.5) -- (0,4.5) node[anchor=east] {$y$};

    % Label x-axis and y-axis
    \foreach \x in {-2,...,8}
        \draw (\x cm,1pt) -- (\x cm,-1pt) node[anchor=north] {$\x$};
    \foreach \y in {-8,...,4}
        \draw (1pt,\y cm) -- (-1pt,\y cm) node[anchor=east] {$\y$};

    % Centre of enlargement
    \coordinate (O) at (0,0);
    \draw[red, thick] (O) node[cross out, draw=red, inner sep=2pt, outer sep=0pt] {};
    \node at (0.4,0.5) {$O$};

    % Original quadrilateral vertices
    \coordinate (A) at (2,-4);
    \coordinate (B) at (2,-8);
    \coordinate (C) at (6,-8);
    \coordinate (D) at (6,-4);

    % Draw the original quadrilateral
    \draw[fill=fillColour, draw=borderColour, line width=1pt] (A) -- (B) -- (C) -- (D) -- cycle;

    \node at (1.6,-4) {$A$};
    \node at (1.6,-8) {$B$};
    \node at (6.4,-8) {$C$};
    \node at (6.4,-4) {$D$};

\end{tikzpicture}
\end{center}

\noindent Enlarge quadrilateral $ABCD$ by scale factor $\dfrac{1}{2}$, using $O$ as the centre of enlargement.

\answerplain{3}
\end{question}
