\begin{question}
\noindent Three drone relay stations, $P$, $Q$ and $R$, are used to broadcast a live e-sports tournament signal across a stadium complex. The stations form a triangle, as shown in the diagram (not drawn accurately).

\noindent $Q$ is due east of $P$.\\
The bearing of $R$ from $P$ is $065^{\circ}$.\\
The bearing of $R$ from $Q$ is $312^{\circ}$.

\begin{center}
\begin{tikzpicture}[scale=1, transform shape]
    \coordinate (P) at (0,0);
    \coordinate (Q) at (6,0);
    \coordinate (R) at (65:6.2);

    \draw[thick] (P) -- (Q) -- (R) -- cycle;

    % North lines at P and Q
    \draw[dashed, ->] (P) -- (0,3.2) node[above]{N};
    \draw[dashed, ->] (Q) -- (6,3.2) node[above]{N};

    % Angle at P between north and PR
    \pic [draw, angle radius=1cm, "$65^{\circ}$", angle eccentricity=1.5] {angle = R--P--(0,3.2)};

    % Angle at Q between north and QR (reflex side, so mark the angle QR to north going clockwise, i.e. 312 means we mark 360-312=48 on the other side)
    \pic [draw, angle radius=1cm, "$48^{\circ}$", angle eccentricity=1.5] {angle = (6,3.2)--Q--R};

    \node at (P) [below left, font=\small] {$P$};
    \node at (Q) [below right, font=\small] {$Q$};
    \node at (R) [above, font=\small] {$R$};
\end{tikzpicture}
\end{center}

\noindent Show that the bearing of $P$ from $R$ is $245^{\circ}$.

\end{question}
