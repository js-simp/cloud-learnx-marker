\begin{question}
\noindent $A$, $B$, $C$ and $D$ are points on the circumference of a circle, forming the cyclic quadrilateral $ABCD$.

\begin{center}
\begin{tikzpicture}[scale=2.2]
    \coordinate[label=above left:$A$] (A) at (130:1);
    \coordinate[label=below left:$D$] (D) at (215:1);
    \coordinate[label=below right:$C$] (C) at (-25:1);
    \coordinate[label=above right:$B$] (B) at (55:1);

    \draw[name path=circ] (0,0) circle (1);
    \draw[line join=bevel] (A) -- (B) -- (C) -- (D) -- cycle;

    \pic [draw, "$3x+10^\circ$", angle eccentricity=1.5, angle radius=0.4cm] {angle = C--D--A};
    \pic [draw, "$2x^\circ$", angle eccentricity=1.5, angle radius=0.4cm] {angle = A--B--C};
\end{tikzpicture}
\end{center}

\noindent Angle $ADC = (3x+10)^\circ$ and angle $ABC = 2x^\circ$.

\begin{enumerate}
    \item[(a)] Find the value of $x$.
    \vspace{3cm}
    \answereq{x}{2}

    \item[(b)] Find the size of angle $ADC$.
    \vspace{2cm}
    \answerequnit{ADC}{$^\circ$}{1}
\end{enumerate}
\end{question}
