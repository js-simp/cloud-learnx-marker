\begin{question}
\noindent $A$, $B$ and $C$ are points on a circle with centre $O$. $AC$ is a diameter of the circle. $AB = BO$.

\begin{center}
\begin{tikzpicture}[scale=2.6]
    \coordinate (O) at (0,0);
    \coordinate (A) at (180:1);
    \coordinate (C) at (0:1);
    \coordinate (B) at (120:1);

    \draw[name path=circ] (0,0) circle (1);
    \fill[black] (O) circle (0.4pt);
    \draw (A) -- (C);
    \draw (A) -- (B);
    \draw (B) -- (C);
    \draw (B) -- (O);

    \node[left] at (A) {$A$};
    \node[right] at (C) {$C$};
    \node[above] at (B) {$B$};
    \node[below] at (O) {$O$};

\end{tikzpicture}
\end{center}

\noindent Given that $AB = BO$, show that angle $BCA = 45^\circ$.
