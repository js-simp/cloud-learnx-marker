\begin{question}
\noindent An architect is designing a memorial garden shaped like an L-shaped path, inspired by the layout of an ancient library described in a historical novel the designer is reading. The composite shape is made from a rectangle with a smaller rectangle removed from one corner, as shown in the diagram.

\begin{center}
\begin{tikzpicture}[scale=0.8]
    % Outer L-shape coordinates
    % Big rectangle would be (0,0) to (3x+4, x+6), with a notch cut from top-right corner
    \draw[thick]
        (0,0) -- (3*1.6+4, 0) -- (3*1.6+4, 1.6+2) -- (1.6+2, 1.6+2) -- (1.6+2, 1.6+6) -- (0, 1.6+6) -- cycle;

    % Labels for sides using algebraic expressions (placed near actual edges, not to scale)
    \node at (1.6*1.5+2, -0.5) {$3x+4$};
    \node at (-1.2, (1.6+6)/2) {$x+6$};
    \node at (1.6+2+0.9, (1.6+2+1.6+6)/2) {$x+4$};
    \node at (1.6+2+1.3, 1.6+2-0.4) {$2x$};
    \node at (1.6/2+2.4, 1.6+2+0.5) {$x+2$};
    \node at (1.6/2-0.8, (1.6+6)+0.5) {$2x+2$};
\end{tikzpicture}
\end{center}

\noindent The diagram shows an L-shaped garden path. All lengths are given in metres, in terms of $x$.

\noindent Show that the perimeter, $P$ metres, of the L-shaped path is given by
\[
P = 10x + 24
\]
\vspace{5cm}
\answermarks{4}
\end{question}
