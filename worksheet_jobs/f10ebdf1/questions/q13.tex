\begin{question}
\noindent A logo designer is creating a circular emblem for a Model United Nations conference badge. Points $A$, $B$, $C$ and $D$ lie on the circumference of a circle with centre $O$, in that order around the circle. Lines $OA$ and $OC$ are drawn to the centre.

Angle $AOC = 136^\circ$ (the angle marked at the centre, on the same side as $B$), and angle $ADC = 118^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2.4]
    \coordinate (O) at (0,0);

    % A at 160 degrees, C needs angle AOC (reflex through B side) = 136 on the B side.
    % Place A at 160, B at 90 (between A and C going the short way through top), C at 24 (160-136=24)
    \coordinate (A) at (160:1);
    \coordinate (B) at (95:1);
    \coordinate (C) at (24:1);
    % D on opposite arc, angle ADC = 118 (D on major arc not containing B)
    \coordinate (D) at (-100:1);

    \draw[name path=circ] (O) circle (1);
    \fill[black] (O) circle (0.8pt);

    \draw[thick] (A) -- (B) -- (C) -- (D) -- cycle;
    \draw[thick] (O) -- (A);
    \draw[thick] (O) -- (C);

    \node[above left] at (A) {$A$};
    \node[above] at (B) {$B$};
    \node[right] at (C) {$C$};
    \node[below] at (D) {$D$};
    \node[below] at (O) {$O$};

    \pic [draw, "$136^\circ$", angle eccentricity=1.5, angle radius=0.5cm] {angle = C--O--A};
    \pic [draw, "$118^\circ$", angle eccentricity=1.5, angle radius=0.55cm] {angle = C--D--A};
\end{tikzpicture}
\end{center}

\begin{enumerate}
    \item[(a)] Find the size of angle $ABC$.\\
    Give a reason for your answer.

    \vspace{4cm}
    \answerequnit{ABC}{$^\circ$}{2}

    \item[(b)] Work out the size of angle $BCD$.\\
    Give a reason for each stage of your working.

    \vspace{4cm}
    \answerequnit{BCD}{$^\circ$}{2}
\end{enumerate}
\end{question}
