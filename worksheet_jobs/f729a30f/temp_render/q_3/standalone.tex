\begin{question}
\noindent An astronomy society is designing a circular observation dome. Points $A$, $B$, $C$ and $D$ lie on the circumference of the circle, representing four telescope mounting positions.

Chord $AB$ is viewed from points $C$ and $D$, which both lie on the same side of $AB$, in the same segment of the circle.

Angle $ACB = 38^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2.2]
    \coordinate (A) at (200:1);
    \coordinate (B) at (340:1);
    \coordinate (C) at (110:1);
    \coordinate (D) at (75:1);

    \draw[thick] (0,0) circle (1);

    \draw[thick] (A) -- (B);
    \draw[thick] (A) -- (C);
    \draw[thick] (C) -- (B);
    \draw[thick] (A) -- (D);
    \draw[thick] (D) -- (B);

    \node[below left] at (A) {$A$};
    \node[below right] at (B) {$B$};
    \node[above left] at (C) {$C$};
    \node[above right] at (D) {$D$};

    \pic [draw, "$38^\circ$", angle eccentricity=1.5, angle radius = 0.45cm] {angle = A--C--B};
    \pic [draw, "$x$", angle eccentricity=1.5, angle radius = 0.45cm] {angle = A--D--B};
\end{tikzpicture}
\end{center}

\noindent Find the size of angle $x$, giving a reason for your answer.
\vspace{4cm}

\begin{flushright}
\answerequnit{x}{$^\circ$}{2}
\end{flushright}
\end{question}
