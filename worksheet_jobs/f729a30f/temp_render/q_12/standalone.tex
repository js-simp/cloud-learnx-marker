\begin{question}
\noindent A particle physics research group models the ring of a circular particle accelerator as a circle. Detectors are positioned at fixed points around the ring to record collision events.

A beam of particles is fired from detector $S$, passes straight through the centre $O$ of the ring, and continues on to detector $U$ on the opposite side. A collision event is also recorded by a third detector, $T$, positioned elsewhere on the ring.

The lines $ST$ and $TU$ are drawn to show the paths analysed by the research group. Angle $TSU = 34^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2.4]
    % Circle centre O
    \coordinate (O) at (0,0);
    \draw[thick] (O) circle (1);
    \fill (O) circle (0.4pt);
    \node[below] at (O) {$O$};

    % S and U are diametrically opposite (beam through centre)
    \coordinate (S) at (200:1);
    \coordinate (U) at (20:1);

    % T is another point on the circle
    \coordinate (T) at (95:1);

    % Draw the beam path S -- O -- U (diameter)
    \draw[thick] (S) -- (U);

    % Draw chords from T
    \draw[thick] (S) -- (T);
    \draw[thick] (T) -- (U);

    % Labels
    \node[left] at (S) {$S$};
    \node[right] at (U) {$U$};
    \node[above] at (T) {$T$};

    % Angle at S (TSU = 34 degrees)
    \pic [draw, "$34^\circ$", angle eccentricity=1.6, angle radius = 0.5cm] {angle = T--S--U};

    % Angle x at T (angle STU) to find
    \pic [draw, "$x$", angle eccentricity=1.5, angle radius = 0.4cm] {angle = U--T--S};

\end{tikzpicture}
\end{center}

\noindent Work out the size of angle $STU$, marked $x$ on the diagram.

\vspace{1cm}
\noindent You must show how you decided which theorem to use.
\vspace{5cm}

\begin{flushright}
\answerequnit{x}{$^\circ$}{4}
\end{flushright}
\end{question}
