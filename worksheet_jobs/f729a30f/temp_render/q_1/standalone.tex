\begin{question}
\noindent $A$, $B$ and $C$ are points on the circumference of a circle with centre $O$.

\noindent Angle $AOC = 100^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2]
    %mark points on circle
        \coordinate[] (O) at (0,0);
        \coordinate[] (A) at (130:1);
        \coordinate[] (C) at (30:1);
        \coordinate[] (B) at (260:1);

    %draw chords and circle
        \draw[name path=circ, fill = orange!10] (0,0) circle (1);
        \fill[black] (0,0) circle (1pt);
        \draw (A) -- (O);
        \draw (O) -- (C);
        \draw (A) -- (B);
        \draw (B) -- (C);

        \node[above left] at (A) {$A$};
        \node[above right] at (C) {$C$};
        \node[below] at (B) {$B$};
        \node[above] at (O) {$O$};

        \pic [draw, "$100^\circ$", angle eccentricity=1.5, angle radius = 0.5cm] {angle = A--O--C};
        \pic [draw, "$x$", angle eccentricity=1.6, angle radius = 0.5cm] {angle = A--B--C};

    \end{tikzpicture}
\end{center}

\noindent Points $A$, $B$ and $C$ lie on the circumference, so angle $ABC$ is the angle subtended at the circumference by the same arc $AC$ that angle $AOC$ is subtended at the centre.

\noindent Work out the size of angle $x$, giving a reason for your answer.
\vspace{4cm}
\begin{flushright}
    \answerequnit{x}{$^\circ$}{2}
\end{flushright}
\end{question}
