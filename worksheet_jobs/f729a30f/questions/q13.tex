\begin{question}
\noindent $PQRS$ is a cyclic quadrilateral, with vertices $P$, $Q$, $R$ and $S$ lying on the circumference of a circle, in that order.

\noindent The interior angles of the quadrilateral are given in terms of $x$:
\[
\angle P = (2x + 5)^\circ, \qquad \angle Q = (3x - 10)^\circ, \qquad \angle R = (x + 15)^\circ, \qquad \angle S = (2x - 5)^\circ
\]

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate[label=above left:$P$] (P) at (130:1);
    \coordinate[label=above right:$Q$] (Q) at (40:1);
    \coordinate[label=below right:$R$] (R) at (-40:1);
    \coordinate[label=below left:$S$] (S) at (215:1);

    \draw[name path=circ] (0,0) circle (1);
    \draw[line join=bevel] (P) -- (Q) -- (R) -- (S) -- cycle;

    \pic [draw, "$(2x+5)^\circ$", angle eccentricity=1.6, angle radius = 0.4cm] {angle = S--P--Q};
    \pic [draw, "$(3x-10)^\circ$", angle eccentricity=1.6, angle radius = 0.4cm] {angle = P--Q--R};
    \pic [draw, "$(x+15)^\circ$", angle eccentricity=1.6, angle radius = 0.4cm] {angle = Q--R--S};
    \pic [draw, "$(2x-5)^\circ$", angle eccentricity=1.6, angle radius = 0.4cm] {angle = R--S--P};
\end{tikzpicture}
\end{center}

\noindent Show that $x = 35$.

    \answermarks{4}
\end{flushright}
\end{question}
