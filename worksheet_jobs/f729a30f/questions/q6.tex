\begin{question}
\noindent Points $A$, $B$ and $C$ lie on a circle with centre $O$.

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate (O) at (0,0);
    \coordinate (A) at (130:1);
    \coordinate (B) at (-30:1);
    \coordinate (C) at (250:1);

    \draw[name path=circ] (0,0) circle (1);
    \fill[black] (0,0) circle (0.4pt);
    \draw (A) -- (O);
    \draw (O) -- (B);
    \draw (A) -- (C);
    \draw (C) -- (B);

    \node[above left] at (A) {$A$};
    \node[right] at (B) {$B$};
    \node[below] at (C) {$C$};
    \node[above] at (O) {$O$};

    \pic [draw, "$(3x+10)^\circ$", angle eccentricity=1.6, angle radius = 0.5cm] {angle = B--O--A};
    \pic [draw, "$(x+30)^\circ$", angle eccentricity=1.6, angle radius = 0.5cm] {angle = B--C--A};
\end{tikzpicture}
\end{center}

\noindent The angle at the centre, $AOB = (3x+10)^\circ$.\\
The angle at the circumference, $ACB = (x+30)^\circ$.

Show that $x = 10$.

\vspace{4.5cm}
\begin{flushright}
\answermarks{3}
\end{flushright}
\end{question}
