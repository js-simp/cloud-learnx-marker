\begin{question}
\noindent $A$, $B$, $C$ and $D$ are points on the circumference of a circle with centre $O$.
$OA$ and $OC$ are radii. Angle $ABC = 64°$.

\begin{center}
\begin{tikzpicture}[scale=2.4]
    \coordinate (O) at (0,0);
    \coordinate (A) at (150:1);
    \coordinate (B) at (250:1);
    \coordinate (C) at (10:1);
    \coordinate (D) at (60:1);

    \draw[name path=circ] (0,0) circle (1);
    \fill[black] (O) circle (0.4pt);
    \node[below] at (O) {$O$};

    \draw (O) -- (A);
    \draw (O) -- (C);
    \draw (A) -- (B) -- (C);
    \draw (A) -- (D) -- (C);

    \node[left] at (A) {$A$};
    \node[below] at (B) {$B$};
    \node[right] at (C) {$C$};
    \node[above] at (D) {$D$};

    \pic [draw, "$64^\circ$", angle eccentricity=1.5, angle radius = 0.35cm] {angle = A--B--C};
    \pic [draw, "$x$", angle eccentricity=1.5, angle radius = 0.5cm] {angle = C--O--A};
    \pic [draw, "$y$", angle eccentricity=1.6, angle radius = 0.4cm] {angle = A--D--C};
\end{tikzpicture}
\end{center}

\noindent $B$ and $D$ lie on opposite arcs $AC$, on either side of the chord $AC$, as shown.

\begin{enumerate}
    \item[(a)] Find the size of angle $x$, the angle at the centre $AOC$ (reflex angle not required — take $x$ as the angle on the same side as $B$).
    \vspace{4cm}
    \answerequnit{x}{$^\circ$}{2}

    \item[(b)] Points $A$, $B$, $C$, $D$ all lie on the circle, with $D$ on the opposite arc to $B$.
    Find the size of angle $y$, angle $ADC$.
    \vspace{3cm}
    \answerequnit{y}{$^\circ$}{2}
\end{enumerate}
\end{question}
