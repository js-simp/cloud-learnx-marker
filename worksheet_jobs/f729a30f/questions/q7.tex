\begin{question}
\noindent $TC$ is a tangent to the circle at point $C$. $A$, $B$ and $C$ are points on the circumference, and $AB$, $BC$ are chords of the circle.

The angle between the tangent $TC$ and the chord $CB$ is $72^\circ$.
The angle between the tangent $TC$ and the chord $CA$ is $115^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2]
    % mark points on circle
    \coordinate (C) at (-90:1);

    % draw circle
    \draw[name path=circ] (0,0) circle (1);

    % draw tangent through C
    \draw (C) -- ($(C)!1.6!-90:(0,0)$) coordinate (L);
    \draw (C) -- ($(C)!1.6!90:(0,0)$) coordinate (Trpt);

    % paths to find B and A on circle
    \path[name path=CB] (C) -- ($(C)!2!-42:(0,0)$);
    \path[name path=CA] (C) -- ($(C)!2!25:(0,0)$);
    \path[name intersections={of=CB and circ, by={temp1,B}}];
    \path[name intersections={of=CA and circ, by={temp2,A}}];

    % draw chords
    \draw (C) -- (B);
    \draw (C) -- (A);
    \draw (A) -- (B);

    % labels
    \node[right] at (Trpt) {$T$};
    \node[below] at (L) {};
    \node[left] at (C) {$C$};
    \node[right] at (B) {$B$};
    \node[above] at (A) {$A$};

    % angle labels
    \pic [draw, "$72^\circ$", angle eccentricity=1.5, angle radius = 0.5cm] {angle = B--C--L};
    \pic [draw, "$115^\circ$", angle eccentricity=1.9, angle radius = 0.7cm] {angle = A--C--L};
    \pic [draw, "$x$", angle eccentricity=1.4, angle radius = 0.5cm] {angle = C--A--B};

\end{tikzpicture}
\end{center}

\noindent Work out the size of angle $x$, marked in the diagram, giving reasons for your answer.
\vspace{4.5cm}
\begin{flushright}
    \answerequnit{x}{$^\circ$}{3}
\end{flushright}
\end{question}
