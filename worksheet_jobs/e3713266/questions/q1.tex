\begin{question}
\noindent $A$, $B$ and $C$ are points on the circumference of a circle, centre $O$.
$AC$ is a diameter of the circle. Angle $BAC = 56^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2.4]
    \coordinate (O) at (0,0);
    \coordinate (A) at (180:1);
    \coordinate (C) at (0:1);
    \coordinate (B) at (60:1);

    \draw (0,0) circle (1);
    \fill[black] (O) circle (0.5pt);
    \node[below] at (O) {$O$};

    \draw[thick] (A) -- (C);
    \draw[thick] (A) -- (B);
    \draw[thick] (B) -- (C);

    \node[left] at (A) {$A$};
    \node[right] at (C) {$C$};
    \node[above] at (B) {$B$};

    \pic [draw, angle radius=6mm, "$56^\circ$"] {angle = C--A--B};
\end{tikzpicture}
\end{center}

\noindent Find the size of angle $ACB$, giving a reason for your answer.

\vspace{4cm}
\answerequnit{ACB}{$^\circ$}{2}
\end{question}
