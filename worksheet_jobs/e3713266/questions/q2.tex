\begin{question}
\noindent $A$, $B$ and $C$ are points on the circumference of a circle with centre $O$.

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate (O) at (0,0);
    \coordinate (A) at (130:1);
    \coordinate (C) at (-30:1);
    \coordinate (B) at (250:1);

    \draw[name path=circ] (0,0) circle (1);
    \fill[black] (0,0) circle (0.5pt);
    \node[below] at (O) {$O$};

    \draw (O) -- (A);
    \draw (O) -- (C);
    \draw (A) -- (B);
    \draw (B) -- (C);

    \node[above left] at (A) {$A$};
    \node[right] at (C) {$C$};
    \node[below] at (B) {$B$};

    \pic [draw, angle eccentricity=1.5, angle radius = 0.5cm, "$116^\circ$"] {angle = C--O--A};
    \pic [draw, angle eccentricity=1.4, angle radius = 0.5cm, "$x$"] {angle = C--B--A};
\end{tikzpicture}
\end{center}

\noindent The angle at the centre, angle $AOC$, is $116^\circ$.

\noindent $B$ lies on the major arc $AC$.

\noindent Find the size of angle $ABC$.

\vspace{4cm}

\begin{flushright}
\answerequnit{ABC}{$^\circ$}{2}
\end{flushright}
\end{question}
