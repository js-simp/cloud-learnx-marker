\begin{question}
\noindent \textbf{(a)} Triangle $P$ has vertices $A(1,1)$, $B(1,2)$ and $C(2,1)$.

Enlarge triangle $P$ by scale factor $2$, centre $(0,0)$.

\begin{center}
\begin{tikzpicture}[scale=0.6]
    \def\axisLimit{6}
    \definecolor{fillColour}{HTML}{D8D8D8}

    % Grid
    \draw[step=1cm,gray,very thin] (-\axisLimit,-\axisLimit) grid (\axisLimit,\axisLimit);

    % Axes
    \draw[thick,->] (-\axisLimit-0.5,0) -- (\axisLimit+0.5,0) node[anchor=north] {$x$};
    \draw[thick,->] (0,-\axisLimit-0.5) -- (0,\axisLimit+0.5) node[anchor=east] {$y$};

    \foreach \x in {-6,...,-1,1,...,6}
        \draw (\x cm,1pt) -- (\x cm,-1pt) node[anchor=north] {$\x$};
    \foreach \y in {-6,...,-1,1,...,6}
        \draw (1pt,\y cm) -- (-1pt,\y cm) node[anchor=east] {$\y$};

    % Centre of enlargement
    \coordinate (COE) at (0,0);
    \draw[red, thick] (COE) node[cross out, draw=red, inner sep=2pt, outer sep=0pt] {};
    \node[below left] at (0,0) {centre};

    % Original triangle P
    \coordinate (A) at (1,1);
    \coordinate (B) at (1,2);
    \coordinate (C) at (2,1);
    \draw[fill=fillColour, draw=black, line width=1pt] (A) -- (B) -- (C) -- cycle;
    \node[left] at (A) {$A$};
    \node[above] at (B) {$B$};
    \node[right] at (C) {$C$};

\end{tikzpicture}
\end{center}

\vspace{4cm}

\answerplain{2}
\end{flushright}
