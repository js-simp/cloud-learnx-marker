\begin{question}
\noindent $A$, $B$, $C$ and $D$ are points on a circle with centre $O$. $TA$ is a tangent to the circle at $A$.

$ABCD$ is a cyclic quadrilateral. Angle $BAT = 3x^\circ$, angle $BCD = (2x + y)^\circ$, and angle $ADC = 4x^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2.4]
    % points on circle
    \coordinate (A) at (200:1);
    \coordinate (B) at (110:1);
    \coordinate (C) at (30:1);
    \coordinate (D) at (-40:1);

    \draw[name path=circ] (0,0) circle (1);

    % tangent at A
    \coordinate (T1) at ($(A)!1.4!90:(0,0)$);
    \coordinate (T2) at ($(A)!0.5!-90:(0,0)$);
    \draw (T2) -- (T1) node[below left]{$T$};

    % chords / quadrilateral
    \draw (A) -- (B) -- (C) -- (D) -- cycle;
    \draw (A) -- (C);

    % labels
    \node at (A) [left]{$A$};
    \node at (B) [above]{$B$};
    \node at (C) [right]{$C$};
    \node at (D) [below right]{$D$};

    \pic [draw, angle eccentricity=1.5, angle radius=0.35cm, "$3x^\circ$"] {angle = T2--A--B};
    \pic [draw, angle eccentricity=1.5, angle radius=0.35cm, "$4x^\circ$"] {angle = C--D--A};
    \pic [draw, angle eccentricity=1.4, angle radius=0.3cm, "$(2x+y)^\circ$"] {angle = D--C--B};
\end{tikzpicture}
\end{center}

\noindent Show that $y = x$.

\vspace{0.3cm}
\noindent You must show each stage of your working, naming any circle theorems you use.
