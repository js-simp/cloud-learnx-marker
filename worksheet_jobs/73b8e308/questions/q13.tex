\begin{question}
\noindent A logo design app displays a circular emblem in which quadrilateral $ABCD$ is inscribed, with vertices in order around the circle. The side $DC$ is extended to an external point $E$, so that $BCE$ is a straight-line exterior angle at $C$.

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate (B) at (130:1);
    \coordinate (A) at (200:1);
    \coordinate (D) at (260:1);
    \coordinate (C) at (330:1);

    \draw[name path=circ] (0,0) circle (1);
    \draw[line join=bevel] (A) -- (B) -- (C) -- (D) -- cycle;

    % extend DC beyond C to E
    \coordinate (E) at ($(C) + 1.1*(C) - 1.1*(D)$);
    \draw (D) -- (E);

    \node at (A) [left] {$A$};
    \node at (B) [above] {$B$};
    \node at (C) [below right] {$C$};
    \node at (D) [below left] {$D$};
    \node at (E) [right] {$E$};

    \pic [draw, "$118^\circ$", angle eccentricity=1.5, angle radius=0.4cm] {angle = D--A--B};
    \pic [draw, "$x$", angle eccentricity=1.5, angle radius=0.45cm] {angle = B--C--E};
\end{tikzpicture}
\end{center}

\noindent Angle $DAB = 118^\circ$.

\noindent Find the size of angle $BCE$, marked $x$ on the diagram, giving full geometric reasons for your method.
\vspace{4.5cm}
\answerequnit{x}{$^\circ$}{3}
\end{question}
