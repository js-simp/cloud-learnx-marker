\begin{question}
\noindent $A$, $B$ and $C$ are points on the circumference of a circle with centre $O$.

\noindent The angle at the centre, $AOC$, is $80^\circ$. The angle at the circumference, $ABC$, is $x^\circ$.

\begin{center}
\begin{tikzpicture}[scale=2]
    \coordinate (O) at (0,0);
    \coordinate (A) at (130:1);
    \coordinate (C) at (50:1);
    \coordinate (B) at (250:1);

    \draw[name path=circ] (0,0) circle (1);
    \fill[black] (O) circle (0.6pt);
    \node at (O) [above] {$O$};

    \draw (A) -- (O) -- (C);
    \draw (A) -- (B) -- (C);

    \node at (A) [above left] {$A$};
    \node at (C) [above right] {$C$};
    \node at (B) [below] {$B$};

    \pic [draw, angle eccentricity=1.5, angle radius = 0.5cm, "$80^\circ$"] {angle = C--O--A};
    \pic [draw, angle eccentricity=1.4, angle radius = 0.5cm, "$x^\circ$"] {angle = C--B--A};
\end{tikzpicture}
\end{center}

\noindent Find the size of angle $x$, giving a reason for your answer.

\vspace{4cm}

\begin{flushright}
\answerequnit{x}{$^\circ$}{2}
\end{flushright}
\end{question}
