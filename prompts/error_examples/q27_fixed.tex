\begin{question}

The radius $OT$ is drawn, and $OP$ is drawn. The angle between the signal line $PT$ and $OP$ is $34^\circ$, as shown in the diagram.

    \coordinate (T) at (-90:1);

    % draw circle
    \draw (0,0) circle (1);
    \fill[black] (0,0) circle (0.6pt);

    % radius
    \draw (O) -- (T);

    % tangent line at T, extended to point P
    \coordinate (P) at ($(T)!2.6!90:(O)$);
    \draw (T) -- (P);

    % line OP
    \draw (O) -- (P);


    % angle at P
    \pic [draw, angle eccentricity=1.5, angle radius=0.8cm, "$34^\circ$"] {angle = T--P--O};

    % labels
    \node at (O) [above] {$O$};
    \node at (T) [below left] {$T$};
    \node at (P) [left] {$P$};
\end{tikzpicture}
\end{center}
