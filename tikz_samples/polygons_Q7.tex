    \pic [draw, "\footnotesize $55^\circ$ ", angle eccentricity=1.5, angle radius=0.5cm] {angle = C--A--B};

    \pic [draw, "\footnotesize $y$", angle eccentricity=1.3, angle radius=0.4cm] {angle = A--C--M};
